# Exercise 14: Practical Search Challenge - Code Navigation
# Task: Navigate this code using search commands
# ──────────────────────────────────────────────────────────────────────────────
class UserService {
    private users: User[] = [];
    
    constructor(private db: Database) {
        this.loadUsers();
    }
    
    async loadUsers(): Promise<void> {
        this.users = await this.db.query('SELECT * FROM users');
    }
    
    getUser(id: number): User | undefined {
        return this.users.find(u => u.id === id);
    }
    
    async createUser(data: UserData): Promise<User> {
        const user = await this.db.insert('users', data);
        this.users.push(user);
        return user;
    }
    
    async updateUser(id: number, data: Partial<UserData>): Promise<User> {
        const user = await this.db.update('users', id, data);
        const index = this.users.findIndex(u => u.id === id);
        this.users[index] = user;
        return user;
    }
    
    async deleteUser(id: number): Promise<void> {
        await this.db.delete('users', id);
        this.users = this.users.filter(u => u.id !== id);
    }
}

Navigation exercises:
  /function   → find functions (or /async for async ones)
  /User       → find User references
  *           → on 'users' to find all usages
  /return     → find return statements
  /this\.db   → find database operations

# Exercise 15: Search in Practical Editing
# Task: Use search to perform bulk edits
# ──────────────────────────────────────────────────────────────────────────────
BEFORE:
const oldName = "value";
const oldPrice = 100;
const oldCount = 5;
function oldFunction() {}
const oldConfig = {};

AFTER (change all "old" to "new"):
Method 1: Search and repeat
  /old     → find first
  cwnew    → change word to "new"
  n.       → repeat

Method 2: Substitute command (bonus!)
  :%s/old/new/g   → replace all in file
  :%s/old/new/gc  → replace all with confirmation

================================================================================
                         SEARCH MOTIONS CHEATSHEET
================================================================================
CHARACTER SEARCH (current line only):
f{char}    - Find {char} forward (cursor ON char)
F{char}    - Find {char} backward (cursor ON char)
t{char}    - Till {char} forward (cursor BEFORE char)
T{char}    - Till {char} backward (cursor AFTER char)
;          - Repeat last f/F/t/T forward
,          - Repeat last f/F/t/T backward

PATTERN SEARCH (whole file):
/{pattern} - Search forward for pattern
?{pattern} - Search backward for pattern
n          - Repeat search in same direction
N          - Repeat search in opposite direction
*          - Search forward for word under cursor
#          - Search backward for word under cursor
g*         - Like * but partial match
g#         - Like # but partial match

SEARCH MODIFIERS:
\c         - Case insensitive: /\cword
\C         - Case sensitive: /\Cword
/pattern/e - Jump to end of match
/pattern/b+n - Jump to beginning + n chars

OPERATORS WITH SEARCH:
d/{pattern} - Delete to pattern
c/{pattern} - Change to pattern
y/{pattern} - Yank to pattern
df{char}    - Delete through {char}
dt{char}    - Delete until {char}
ct{char}    - Change until {char}

USEFUL SETTINGS:
:set hlsearch   - Highlight all matches
:set incsearch  - Show matches as you type
:noh            - Clear search highlighting
